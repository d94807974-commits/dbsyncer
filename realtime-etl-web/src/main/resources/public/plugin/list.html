  3. NewPluginsCategory

    New Category，Like what? MyPlugin，Achieve interface PluginService Methodology

    package org.test;

import org.realtimeetl.sdk.connector.ConnectorInstance;
import org.realtimeetl.sdk.connector.database.DatabaseConnectorInstance;
import org.realtimeetl.sdk.connector.database.ds.SimpleConnection;
import org.realtimeetl.sdk.plugin.PluginContext;
import org.realtimeetl.sdk.spi.PluginService;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class MyPlugin implements PluginService {

    private final Logger logger = LoggerFactory.getLogger(getClass());

    /**
 * FullSync/TimeIncrementalSync Preprocessing
     *
 * @param pluginContext Context
     */
    @Override
    public void postProcessBefore(PluginContext pluginContext) {
    }

    /**
 * FullSync/IncrementalSync
     *
     * @param pluginContext
     */
    @Override
    public void convert(PluginContext pluginContext) {
 // Expand Moving Data Functions
 logger.info("PluginsExpanding conversion data...");

 // Whether to stop writing to the target DB; default false
        pluginContext.setTerminated(false);

 // SourceTableandTargetTable
        pluginContext.getSourceTableName();
        pluginContext.getTargetTableName();

 // Capture events（INSERT/UPDATE/DELETE）
        pluginContext.getEvent();

 // SourceandTargetTableFullorIncrementalData
        pluginContext.getSourceList();
        pluginContext.getTargetList();

 // AccessTarget databaseConnectorExamples（If necessaryConnector，Must be introducedrealtime-etl-connector-[[${version}]].jar）
        pluginContext.getTargetConnectorInstance();
    }

    /**
 * FullSync/IncrementalSynchronize processing
     *
     * @param context
     */
    @Override
    public void postProcessAfter(PluginContext context) {
 // Call this method after sync
 logger.info("PluginsProcessingSync succeededData，TargetTable:{}，Events:{}，Number Mediation:{}", context.getTargetTableName(),
                        context.getEvent(), context.getTargetList().size());

        ConnectorInstance connectorInstance = context.getSourceConnectorInstance();

 // Access relationship databaseConnections，Achieving its business logic...
        if (connectorInstance instanceof DatabaseConnectorInstance) {
            DatabaseConnectorInstance db = (DatabaseConnectorInstance) connectorInstance;
 // Mode I：
            String query = "select * from my_user limit 10";
            db.execute(databaseTemplate -> databaseTemplate.queryForList(query));

 // Mode XVII，Commence Concerning：
            int execute = db.execute(databaseTemplate -> {
                SimpleConnection connection = databaseTemplate.getSimpleConnection();
                int update = 0;
                try {
 // ManualSubmitService
                    connection.setAutoCommit(false);
                    Object[] args = new Object[3];
                    args[0] = "1";
                    args[1] = "2";
                    args[2] = "3";
                    update = databaseTemplate.update("INSERT INTO table_name (column1, column2,
                        column3) VALUES (value1, value2, value3)", args);
                    connection.commit();
                } catch (Exception e) {
 // Unusual Rollback
                    connection.rollback();
                } finally {
 // Revert DefaultSubmitService
                    connection.setAutoCommit(true);
                }
                return update;
            });
 logger.info("Manual executionSQLResult:{}", execute);
        }
    }

    /**
 * Rewrite Method：Set Version Number
     *
     * @return
     */
    @Override
    public String getVersion() {
        return "1.0.0";
    }

    /**
 * Rewrite Method：SettingsPluginsName
     *
     * @return
     */
    @Override
    public String getName() {
 return "Mine.Plugins";
    }
}
