How to develop plugins?

The plugin is a technology that allows the full and incremental synchronization of data conversion. The plugin allows for the reception of synchronized data, customizes synchronized line data to the target source, and can consume data and achieve more business scenarios.

  1. 7. New Project

    New java or maven project, or download Example items

  2. Import Development Package

    Mode 1 (recommended): Introduction pom (required locally)

    <dependency>
    <groupId>org.ghi</groupId>
    <artifactId>dbsyncer-sdk</artifactId>
    <version>[[${version}]]</version>
</dependency>

    Mode 2: Import jar

    dbsyncer-sdk-[[${version}]].jar

  3. New Plugin Class

    Create a new class, such as MyPlugin, to achieve the PluginService method

    package org.test;

import org.dbsyncer.sdk.connector.ConnectorInstance;
import org.dbsyncer.sdk.connector.database.DatabaseConnectorInstance;
import org.dbsyncer.sdk.connector.database.ds.SimpleConnection;
import org.dbsyncer.sdk.plugin.PluginContext;
import org.dbsyncer.sdk.spi.PluginService;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class MyPlugin implements PluginService {

    private final Logger logger = LoggerFactory.getLogger(getClass());

    /**
     * Pre-process before full sync / scheduled incremental sync
     *
     * @param pluginContext context
     */
    @Override
    public void postProcessBefore(PluginContext pluginContext) {
    }

    /**
     * Full sync / incremental sync
     *
     * @param pluginContext
     */
    @Override
    public void convert(PluginContext pluginContext) {
        // Extend data conversion
        logger.info("Plugin is converting data...");

        // Whether to abort writing to the target (default false)
        pluginContext.setTerminated(false);

        // Source and target tables
        pluginContext.getSourceTableName();
        pluginContext.getTargetTableName();

        // Captured event (INSERT/UPDATE/DELETE)
        pluginContext.getEvent();

        // Full or incremental rows from source/target tables
        pluginContext.getSourceList();
        pluginContext.getTargetList();

        // Get target connector instance (if needed, add dbsyncer-connector-[[${version}]].jar)
        pluginContext.getTargetConnectorInstance();
    }

    /**
     * Post-process after full/incremental sync completes
     *
     * @param context
     */
    @Override
    public void postProcessAfter(PluginContext context) {
        // Called after sync completes
        logger.info("Plugin handling synced data, target table:{}, event:{}, count:{}", context.getTargetTableName(),
                        context.getEvent(), context.getTargetList().size());

        ConnectorInstance connectorInstance = context.getSourceConnectorInstance();

        // Obtain a relational DB connection and implement custom logic...
        if (connectorInstance instanceof DatabaseConnectorInstance) {
            DatabaseConnectorInstance db = (DatabaseConnectorInstance) connectorInstance;
            // Option 1:
            String query = "select * from my_user limit 10";
            db.execute(databaseTemplate -> databaseTemplate.queryForList(query));

            // Option 2, with transaction:
            int execute = db.execute(databaseTemplate -> {
                SimpleConnection connection = databaseTemplate.getSimpleConnection();
                int update = 0;
                try {
                    // ManualSubmitService
                    connection.setAutoCommit(false);
                    Object[] args = new Object[3];
                    args[0] = "1";
                    args[1] = "2";
                    args[2] = "3";
                    update = databaseTemplate.update("INSERT INTO table_name (column1, column2,
                        column3) VALUES (value1, value2, value3)", args);
                    connection.commit();
                } catch (Exception e) {
                    // ErrorRoll back
                    connection.rollback();
                } finally {
                    // Revert DefaultSubmitService
                    connection.setAutoCommit(true);
                }
                return update;
            });
            logger.info("Manual Transition", execute);
        }
    }

    /**
     * Rewrite method:SettingsVersionNo. No.
     *
     * @return
     */
    @Override
    public String getVersion() {
        return "1.0.0";
    }

    /**
     * Rewrite method:SettingsPluginName
     *
     * @return
     */
    @Override
    public String getName() {
        return "My Plugin";
    }
}
